cmake_minimum_required(VERSION 3.24)
project(imh_exr VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenImageIO CONFIG REQUIRED)
find_package(OpenColorIO CONFIG REQUIRED)
find_package(OpenEXR CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(imh-exr main.cpp)
target_link_libraries(imh-exr PRIVATE OpenImageIO::OpenImageIO OpenColorIO::OpenColorIO OpenEXR::OpenEXR nlohmann_json::nlohmann_json)
install(TARGETS imh-exr RUNTIME DESTINATION .)
if(WIN32)
  install(FILES $<TARGET_RUNTIME_DLLS:imh-exr> DESTINATION .)
endif()
enable_testing()
add_test(NAME protocol-self-test COMMAND imh-exr --test)
